cmake_minimum_required(VERSION 3.16)
project(local_libavoid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LIBAVOID_BUILD_SIMULINK_EXT "Build local router" ON)
option(LIBAVOID_BUILD_TESTS "Build functional tests and benchmarks" ON)
option(LIBAVOID_BUILD_LEGACY_TESTS "Build five native smoke tests" ON)
add_library(avoid STATIC actioninfo.cpp connectionpin.cpp connector.cpp connend.cpp geometry.cpp
  geomtypes.cpp graph.cpp hyperedge.cpp hyperedgeimprover.cpp hyperedgetree.cpp junction.cpp
  makepath.cpp mtst.cpp obstacle.cpp orthogonal.cpp router.cpp scanline.cpp shape.cpp timer.cpp
  vertices.cpp viscluster.cpp visibility.cpp vpsc.cpp)
target_include_directories(avoid PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_definitions(avoid PUBLIC LIBAVOID_NO_DLL)
enable_testing()
if(LIBAVOID_BUILD_LEGACY_TESTS)
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/output)
  foreach(t example connectionpin01 inlineShapes checkpoints01 overlappingRects)
    add_executable(${t} tests/${t}.cpp)
    target_link_libraries(${t} PRIVATE avoid)
    add_test(NAME native.${t} COMMAND ${t})
  endforeach()
endif()
if(LIBAVOID_BUILD_SIMULINK_EXT)
  add_subdirectory(simulink_ext)
endif()
